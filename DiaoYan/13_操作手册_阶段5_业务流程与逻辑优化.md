```dart
// lib/core/error/app_exception.dart
/// 应用统一异常体系
/// 所有业务层异常均继承自此，便于上层统一捕获、上报、降级

abstract class AppException implements Exception {
  const AppException(this.message, [this.code, this.details]);
  
  final String message;
  final String? code;       // 错误码，用于国际化/日志分类
  final Map<String, dynamic>? details; // 附加上下文
  
  @override
  String toString() => '$runtimeType($code): $message';
}

/// 网络层异常
class NetworkException extends AppException {
  const NetworkException(super.message, {super.code, super.details, this.statusCode});
  final int? statusCode;
  
  // 常用预定义
  static const NetworkException timeout = NetworkException(
    'Request timeout', code: 'NET_TIMEOUT', statusCode: 408);
  static const NetworkException offline = NetworkException(
    'No internet connection', code: 'NET_OFFLINE', statusCode: 0);
}

/// LLM 调用异常
class LLMException extends AppException {
  const LLMException(super.message, {super.code, super.details, 
    this.provider, this.model, this.retryable = true});
  final String? provider;
  final String? model;
  final bool retryable;
  
  static LLMException fromDioError(DioException e, {String? provider, String? model}) {
    switch (e.type) {
      case DioExceptionType.connectionTimeout:
      case DioExceptionType.receiveTimeout:
        return LLMException('LLM request timeout', code: 'LLM_TIMEOUT', 
          provider: provider, model: model, retryable: true);
      case DioExceptionType.badResponse:
        final status = e.response?.statusCode ?? 500;
        return LLMException('LLM error: $status', code: 'LLM_HTTP_$status',
          provider: provider, model: model, retryable: status >= 500);
      default:
        return LLMException('LLM request failed: ${e.message}', code: 'LLM_FAILED',
          provider: provider, model: model, retryable: false);
    }
  }
}

/// 数据库异常
class DatabaseException extends AppException {
  const DatabaseException(super.message, {super.code, super.details, this.query});
  final String? query;
}

/// 解析异常
class ParseException extends AppException {
  const ParseException(super.message, {super.code, super.details, this.rawInput});
  final String? rawInput;
}

/// 业务逻辑异常（参数校验、状态非法等）
class BusinessException extends AppException {
  const BusinessException(super.message, {super.code, super.details});
  
  static const BusinessException invalidState = BusinessException(
    'Invalid state for operation', code: 'BIZ_INVALID_STATE');
  static const BusinessException notFound = BusinessException(
    'Resource not found', code: 'BIZ_NOT_FOUND');
  static const BusinessException permissionDenied = BusinessException(
    'Permission denied', code: 'BIZ_PERMISSION_DENIED');
}

/// 取消操作异常（用于 GenerationController）
class CancellationException extends AppException {
  const CancellationException() : super('Operation cancelled', code: 'CANCELLED');
}

/// 统一错误处理器
class ErrorHandler {
  static void handleError(Object error, StackTrace stack, {String? context}) {
    // 1. 记录日志
    KiraLogger().error(context ?? 'ErrorHandler', error.toString(), stack);
    
    // 2. 上报崩溃收集（可选：Sentry、Firebase Crashlytics）
    // _reportToCrashlytics(error, stack, context);
    
    // 3. 业务降级策略
    if (error is LLMException && error.retryable) {
      // 触发重试逻辑
    }
  }
  
  /// 安全执行，自动捕获并转换为 Result
  static Future<Result<T>> safeCall<T>(Future<T> Function() fn, {String? context}) async {
    try {
      final data = await fn();
      return Result.success(data);
    } catch (e, stack) {
      handleError(e, stack, context: context);
      return Result.failure(e is AppException ? e : AppException(e.toString()));
    }
  }
}

/// 结果封装（避免异常穿透业务逻辑）
class Result<T> {
  final T? _data;
  final AppException? _error;
  
  const Result._success(this._data) : _error = null;
  const Result._failure(this._error) : _data = null;
  
  factory Result.success(T data) => Result._success(data);
  factory Result.failure(AppException error) => Result._failure(error);
  
  bool get isSuccess => _error == null;
  bool get isFailure => _error != null;
  
  T get data => _data ?? (throw StateError('Result is failure'));
  AppException get error => _error ?? (throw StateError('Result is success'));
  
  T unwrapOr(T defaultValue) => isSuccess ? data : defaultValue;
  T unwrapOrElse(T Function(AppException) fn) => isSuccess ? data : fn(error);
}
```

### 步骤 5.2：补全群聊流程异常分支（P0，B1）

**目标文件**：`lib/presentation/providers/chat/generation_controller.dart`（阶段 4 产出）

**当前缺失**：`GroupSettings.responseMode` 5 种模式下的异常处理

**补全内容**：

```dart
// generation_controller.dart - 群聊生成逻辑增强

Future<void> _generateGroupResponse(ChatMessage userMessage) async {
  final group = _state.group!;
  final members = _state.groupCharacters.values.toList();
  final mode = group.settings.responseMode;
  
  final responders = _selectResponders(members, mode);
  
  for (final character in responders) {
    final token = CancellationToken();
    _activeTokens[character.id] = token;
    
    try {
      await _generateSingleCharacterResponse(
        character: character,
        userMessage: userMessage,
        token: token,
        onChunk: (chunk) => _appendToCharacterMessage(character.id, chunk),
      );
    } on LLMException catch (e) {
      // 单角色失败：记录错误、发送系统提示、继续下一角色
      KiraLogger().warning('GroupChat', 'Character ${character.name} generation failed: $e');
      await _sendSystemMessage('⚠️ ${character.name} 生成失败: ${e.message}，已跳过');
      // 根据配置决定是否停止整个群聊
      if (group.settings.stopOnError) break;
    } on CancellationException {
      // 被取消：清理状态、退出循环
      break;
    } catch (e, stack) {
      // 未知错误：上报、发送通用提示、继续
      ErrorHandler.handleError(e, stack, context: 'GroupChat.${character.name}');
      await _sendSystemMessage('⚠️ ${character.name} 发生未知错误，已跳过');
    } finally {
      _activeTokens.remove(character.id);
    }
  }
  
  // 全部完成后更新状态
  _state = _state.copyWith(isGenerating: false, currentResponderId: null);
}

List<Character> _selectResponders(List<Character> members, GroupResponseMode mode) {
  switch (mode) {
    case GroupResponseMode.sequential:
      return members;
    case GroupResponseMode.random:
      return List.from(members)..shuffle();
    case GroupResponseMode.roundRobin:
      // 基于上轮发言者轮询
      final lastResponder = _state.messages.lastWhere(
        (m) => m.role == 'assistant',
        orElse: () => null,
      );
      if (lastResponder != null) {
        final idx = members.indexWhere((c) => c.id == lastResponder.characterId);
        return [...members.sublist(idx + 1), ...members.sublist(0, idx + 1)];
      }
      return members;
    case GroupResponseMode.parallel:
      return members; // 并行由调用者处理
    case GroupResponseMode.specific:
      final targetId = group.settings.specificCharacterId;
      return members.where((c) => c.id == targetId).toList();
  }
}
```

```dart
// context_step.dart
abstract class ContextStep {
  /// 步骤名称（用于日志/调试）
  String get name;
  
  /// 执行步骤
  /// [ctx] 包含所有上游步骤产出的数据
  /// 返回修改后的 messages 或 null 表示不修改
  Future<List<ChatMessage>?> execute(ContextData ctx);
  
  /// 是否启用（可由配置控制）
  bool get enabled => true;
  
  /// 执行优先级（数值小先执行）
  int get priority => 100;
}

// context_pipeline.dart
class ContextPipeline {
  final List<ContextStep> _steps = [];
  
  void addStep(ContextStep step) {
    _steps.add(step);
    _steps.sort((a, b) => a.priority.compareTo(b.priority));
  }
  
  Future<List<ChatMessage>> run(ContextData initial) async {
    var messages = initial.messages;
    var ctx = initial.copyWith(messages: messages);
    
    for (final step in _steps) {
      if (!step.enabled) continue;
      
      final stopwatch = Stopwatch()..start();
      try {
        final result = await step.execute(ctx);
        if (result != null) {
          messages = result;
          ctx = ctx.copyWith(messages: messages);
        }
        KiraLogger().debug('ContextPipeline', '${step.name} took ${stopwatch.elapsedMs}ms');
      } catch (e, stack) {
        ErrorHandler.handleError(e, stack, context: 'ContextPipeline.${step.name}');
        // 步骤失败不阻断管道，记录错误继续
      }
    }
    
    return messages;
  }
}

// context_builder.dart
class ContextBuilder {
  ContextBuilder({
    required this.worldBookMatcher,
    required this.variablesService,
    required this.macroService,
    required this.promptManager,
    required this.personaService,
  });
  
  final WorldBookInjectionService worldBookMatcher;
  final VariablesService variablesService;
  final MacroService macroService;
  final PromptManagerService promptManager;
  final PersonaService personaService;
  
  Future<List<ChatMessage>> build({
    required List<ChatMessage> messages,
    required Character character,
    required Group? group,
    required Persona? persona,
    required LLMConfig config,
    required ChatSettings settings,
  }) async {
    final pipeline = ContextPipeline();
    
    // 注册标准步骤（按优先级）
    pipeline.addStep(WorldBookStep(worldBookMatcher, character, settings));
    pipeline.addStep(VariableStep(variablesService, settings));
    pipeline.addStep(MacroStep(macroService));
    pipeline.addStep(PromptStep(promptManager, character, settings));
    pipeline.addStep(PersonaStep(personaService, persona));
    pipeline.addStep(SystemPromptStep(character, config, settings));
    
    // 执行管道
    return await pipeline.run(ContextData(
      messages: messages,
      character: character,
      group: group,
      persona: persona,
      config: config,
      settings: settings,
    ));
  }
}
```

```dart
// steps/world_book_step.dart
class WorldBookStep extends ContextStep {
  @override
  String get name => 'WorldBookInjection';
  
  @override
  int get priority => 10;
  
  WorldBookStep(this._matcher, this._character, this._settings);
  
  final WorldBookInjectionService _matcher;
  final Character _character;
  final ChatSettings _settings;
  
  @override
  Future<List<ChatMessage>?> execute(ContextData ctx) async {
    if (!_settings.enableWorldBook) return null;
    
    final entries = await _matcher.inject(
      messages: ctx.messages,
      character: _character,
      scanDepth: _settings.worldBookScanDepth,
    );
    
    if (entries.isEmpty) return null;
    
    // 将注入条目转换为 system 消息插入
    final injectionMessages = _buildInjectionMessages(entries, _settings);
    return _insertAtDepth(ctx.messages, injectionMessages, _settings.worldBookDepth);
  }
}

// steps/variable_step.dart
class VariableStep extends ContextStep {
  @override
  String get name => 'VariableSubstitution';
  
  @override
  int get priority => 20;
  
  VariableStep(this._service, this._settings);
  
  final VariablesService _service;
  final ChatSettings _settings;
  
  @override
  Future<List<ChatMessage>?> execute(ContextData ctx) async {
    if (!_settings.enableVariables) return null;
    
    var messages = ctx.messages;
    for (int i = 0; i < messages.length; i++) {
      final processed = await _service.processMessage(
        messages[i].content,
        chatId: ctx.chatId,
        characterId: ctx.character?.id,
      );
      if (processed != messages[i].content) {
        messages = [...messages];
        messages[i] = messages[i].copyWith(content: processed);
      }
    }
    return messages;
  }
}
```

### 步骤 5.4：发送消息幂等性与重试（P1，B4）

**问题**：网络抖动导致重复发送、或发送失败无重试。

**方案**：客户端生成 `messageId`（UUID），服务端去重；Provider 层实现指数退避重试。

```dart
// chat_notifier.dart
class ActiveChatNotifier extends AutoDisposeNotifier<ActiveChatState> {
  // 消息去重缓存（内存 + 持久化）
  final _sentMessageIds = <String, DateTime>{};
  
  Future<void> sendMessage(String text, List<ChatAttachment> attachments) async {
    final messageId = Uuid().v4();
    final now = DateTime.now();
    
    // 幂等性检查：5 秒内相同内容视为重复
    final recentDuplicate = _sentMessageIds.entries
        .where((e) => now.difference(e.value).inSeconds < 5)
        .any((e) => _isSimilarContent(e.key, text));
    if (recentDuplicate) {
      KiraLogger().warning('Chat', 'Duplicate message detected, ignoring');
      return;
    }
    
    _sentMessageIds[messageId] = now;
    
    // 带重试的发送
    await _sendWithRetry(messageId, text, attachments, maxRetries: 3);
  }
  
  Future<void> _sendWithRetry(
    String messageId, 
    String text, 
    List<ChatAttachment> attachments,
    {int maxRetries = 3}
  ) async {
    for (int attempt = 0; attempt <= maxRetries; attempt++) {
      try {
        await _generationController.sendMessage(text, attachments, messageId: messageId);
        return; // 成功
      } on NetworkException catch (e) {
        if (attempt == maxRetries) rethrow;
        final delay = Duration(seconds: 1 << attempt); // 1s, 2s, 4s
        KiraLogger().info('Chat', 'Send failed, retry in ${delay.inSeconds}s: $e');
        await Future.delayed(delay);
      }
    }
  }
}
```

### 步骤 5.6：世界书注入递归保护（P1，B6）

**现状**：`recursionDepth` 配置但无循环检测。

**增强**：

```dart
// world_book_injection_service.dart
class WorldBookInjectionService {
  // 线程安全的递归追踪
  final _recursionStack = <String, int>{};
  final _lock = Lock();
  
  Future<List<InjectedEntry>> inject({
    required List<ChatMessage> messages,
    required Character character,
    int scanDepth = 10,
    int recursionDepth = 3,
  }) async {
    final visited = <String>{}; // 已注入条目 ID
    final results = <InjectedEntry>[];
    
    await _scanRecursive(
      messages: messages,
      character: character,
      currentDepth: 0,
      maxDepth: recursionDepth,
      visited: visited,
      results: results,
      scanDepth: scanDepth,
    );
    
    return results;
  }
  
  Future<void> _scanRecursive({
    required List<ChatMessage> messages,
    required Character character,
    required int currentDepth,
    required int maxDepth,
    required Set<String> visited,
    required List<InjectedEntry> results,
    required int scanDepth,
  }) async {
    if (currentDepth >= maxDepth) return;
    
    final entries = await _matcher.matchKeywords(
      messages: messages.take(scanDepth).toList(),
      character: character,
    );
    
    for (final entry in entries) {
      if (visited.contains(entry.id)) continue; // 防循环
      visited.add(entry.id);
      results.add(entry);
      
      // 递归扫描注入内容本身的关键词
      if (entry.recursive) {
        final injectionMessages = [
          ChatMessage.system(content: entry.content),
        ];
        await _scanRecursive(
          messages: injectionMessages,
          character: character,
          currentDepth: currentDepth + 1,
          maxDepth: maxDepth,
          visited: visited,
          results: results,
          scanDepth: scanDepth,
        );
      }
    }
  }
}
```

```dart
  global,      // 应用级，跨聊天持久化
  chat,        // 聊天级，随聊天持久化
  character,   // 角色级，随角色持久化
  temp,        // 临时，仅本轮生成有效
}

class VariablesService {
  // 作用级存储
  final Map<VariableScope, Map<String, dynamic>> _stores = {
    VariableScope.global: {},
    VariableScope.chat: {},
    VariableScope.character: {},
    VariableScope.temp: {},
  };
  
  /// 获取变量（按优先级：temp > character > chat > global）
  dynamic get(String name, {VariableScope? scope}) {
    if (scope != null) {
      return _stores[scope]?[name];
    }
    for (final s in [VariableScope.temp, VariableScope.character, VariableScope.chat, VariableScope.global]) {
      if (_stores[s]!.containsKey(name)) return _stores[s]![name];
    }
    return null;
  }
  
  /// 设置变量
  void set(String name, dynamic value, VariableScope scope) {
    _stores[scope]![name] = value;
    _persist(scope);
  }
  
  /// 清除作用域（聊天结束时）
  void clearScope(VariableScope scope) {
    _stores[scope]!.clear();
    _persist(scope);
  }
}
```

### 步骤 5.10：状态更新批量化（P2，B10）

**问题**：`chat_providers.dart` 中频繁 `copyWith` 触发大量重建。

**优化**：批量累积状态变更，单帧内合并一次 `notifyListeners`。

```dart
// generation_controller.dart
class GenerationController extends ChangeNotifier {
  final _pendingUpdates = <VoidCallback>[];
  Timer? _flushTimer;
  
  void _scheduleUpdate(VoidCallback update) {
    _pendingUpdates.add(update);
    _flushTimer ??= Timer(const Duration(milliseconds: 16), _flushUpdates);
  }
  
  void _flushUpdates() {
    _flushTimer = null;
    if (_pendingUpdates.isEmpty) return;
    
    // 合并执行
    for (final update in _pendingUpdates) {
      update();
    }
    _pendingUpdates.clear();
    notifyListeners(); // 仅通知一次
  }
  
  // 使用示例
  void _onChunkReceived(LLMStreamChunk chunk) {
    _scheduleUpdate(() {
      _currentMessage = _currentMessage.copyWith(
        content: _currentMessage.content + (chunk.content ?? ''),
        reasoning: (_currentMessage.reasoning ?? '') + (chunk.reasoning ?? ''),
      );
    });
  }
}
```
