/// 初始化模块：封装所有启动时的依赖创建与 Provider 覆盖
class InitializationModule {
  final SharedPreferences prefs;
  final AppDatabase database;
  final String dataPath;
  
  // Repositories
  late final CharacterRepository characterRepository;
  late final ChatRepository chatRepository;
  late final WorldInfoRepository worldInfoRepository;
  late final PersonaRepository personaRepository;
  late final GroupRepository groupRepository;
  late final TagRepository tagRepository;
  late final LLMConfigRepository llmConfigRepository;
  late final ApiCredentialRepository apiCredentialRepository;
  late final BookmarkRepository bookmarkRepository;
  
  // Services
  late final EJSRenderRegistry ejsRegistry;
  late final LLMService llmService;
  late final ImportService importService;
  
  // Provider overrides
  late final List<Override> overrides;

  InitializationModule._(this.prefs, this.database, this.dataPath) {
    _createRepositories();
    _createServices();
    _buildOverrides();
  }

  /// 工厂方法：完整初始化（用于 main.dart）
  static Future<InitializationModule> create() async {
    final prefs = await SharedPreferences.getInstance();
    final initData = await InitializationService.initialize();
    return InitializationModule._(prefs, initData.database, initData.dataPath);
  }

  /// 工厂方法：测试用（可注入 Mock）
  static InitializationModule forTesting({
    required SharedPreferences prefs,
    required AppDatabase database,
    required String dataPath,
  }) {
    return InitializationModule._(prefs, database, dataPath);
  }

  void _createRepositories() {
    characterRepository = CharacterRepository(database, dataPath);
    chatRepository = ChatRepository(database);
    worldInfoRepository = WorldInfoRepository(database);
    personaRepository = PersonaRepository(database);
    groupRepository = GroupRepository(database);
    tagRepository = TagRepository(database);
    llmConfigRepository = LLMConfigRepository(database);
    apiCredentialRepository = ApiCredentialRepository(database);
    bookmarkRepository = BookmarkRepository(database);
  }

  void _createServices() {
    ejsRegistry = EJSRenderRegistry();
    llmService = LLMService(ejsRenderer: RegistryEJSRenderer(ejsRegistry));
    importService = ImportService(dataPath);
  }

  void _buildOverrides() {
    overrides = [
      // Database
      databaseProvider.overrideWithValue(database),
      // Repositories
      characterRepositoryProvider.overrideWithValue(characterRepository),
      chatRepositoryProvider.overrideWithValue(chatRepository),
      worldInfoRepositoryProvider.overrideWithValue(worldInfoRepository),
      personaRepositoryProvider.overrideWithValue(personaRepository),
      groupRepositoryProvider.overrideWithValue(groupRepository),
      tagRepositoryProvider.overrideWithValue(tagRepository),
      llmConfigRepositoryProvider.overrideWithValue(llmConfigRepository),
      apiCredentialRepositoryProvider.overrideWithValue(apiCredentialRepository),
      bookmarkRepositoryProvider.overrideWithValue(bookmarkRepository),
      // Services
      llmServiceProvider.overrideWithValue(llmService),
      ejsRenderRegistryProvider.overrideWithValue(ejsRegistry),
      importServiceProvider.overrideWithValue(importService),
      // Shared preferences
      sharedPreferencesProvider.overrideWithValue(prefs),
    ];
  }
}

import 'dart:io';

void main() async {
  final providersDir = Directory('lib/presentation/providers');
  final files = providersDir.listSync().whereType<File>().where((f) => f.path.endsWith('_providers.dart'));
  
  final buffer = StringBuffer();
  buffer.writeln('// ignore_for_file: library_private_types_in_public_api');
  buffer.writeln('// 此文件由脚本生成，勿手动编辑');
  buffer.writeln('// 生成命令: dart run scripts/generate_provider_registry.dart');
  buffer.writeln('');
  buffer.writeln('/// Provider 注册清单');
  buffer.writeln('///');
  buffer.writeln('/// 分类对应 `lib/presentation/providers/` 目录下文件');
  buffer.writeln('');
  
  int total = 0;
  for (final file in files) {
    final content = await file.readAsString();
    final providers = RegExp(r'final\s+(\w+Provider)\s*=').allMatches(content);
    if (providers.isNotEmpty) {
      final category = file.path.split('/').last.replaceAll('_providers.dart', '');
      buffer.writeln('// $category (${providers.length} providers)');
      buffer.writeln("export '${file.path.replaceAll('\\\\', '/').replaceFirst('lib/', 'package:kirakira/')}' show");
      for (final m in providers) {
        buffer.writeln('    ${m.group(1)},');
        total++;
      }
      buffer.writeln(';');
      buffer.writeln('');
    }
  }
  
  buffer.writeln('/// 统计信息');
  buffer.writeln('const int _totalProviders = $total;');
  buffer.writeln("const String _generatedAt = '${DateTime.now().toIso8601String().split('T').first}';");
  
  await File('lib/presentation/providers/providers_registry.dart').writeAsString(buffer.toString());
  print('Generated providers_registry.dart with $total providers');
}

### 步骤 1.4：依赖方向校验（架构守护）

**目标**：建立自动化检查，防止未来引入循环依赖或逆向依赖。

**操作文件**：
- 新建：`scripts/verify_architecture.dart`

**详细操作**：

```dart
// scripts/verify_architecture.dart
import 'dart:io';

/// 允许的依赖方向：Presentation → Domain → Data → Core
/// 禁止：反向、同层跨模块、Core → 任意上层
final allowedDirections = {
  'presentation': ['domain', 'data', 'core'],
  'domain': ['data', 'core'],
  'data': ['core'],
  'core': [],
};

final layerOf = {
  'presentation': 'presentation',
  'domain': 'domain',
  'data': 'data',
  'core': 'core',
  'services': 'presentation',  // lib/services/ 归属 presentation
  'widgets': 'presentation',   // lib/widgets/ 归属 presentation
};

void main() async {
  final libDir = Directory('lib');
  final dartFiles = <File>[];
  await for (final entity in libDir.list(recursive: true)) {
    if (entity is File && entity.path.endsWith('.dart')) {
      dartFiles.add(entity);
    }
  }
  
  int violations = 0;
  for (final file in dartFiles) {
    final content = await file.readAsString();
    final imports = RegExp(r"import\s+['\"](package:kirakira/([^'\"]+))['\"];")
        .allMatches(content);
    
    final currentLayer = _getLayer(file.path);
    if (currentLayer == null) continue;
    
    for (final m in imports) {
      final importPath = m.group(2)!;
      final targetLayer = _getLayerFromImport(importPath);
      if (targetLayer == null) continue;
      
      if (!allowedDirections[currentLayer]!.contains(targetLayer)) {
        print('❌ 违规依赖: $file (layer: $currentLayer) → $importPath (layer: $targetLayer)');
        violations++;
      }
    }
  }
  
  if (violations == 0) {
    print('✅ 架构依赖检查通过：无违规依赖');
    exit(0);
  } else {
    print('⚠️ 发现 $violations 个违规依赖');
    exit(1);
  }
}

String? _getLayer(String path) {
  for (final layer in ['presentation', 'domain', 'data', 'core', 'services', 'widgets']) {
    if (path.contains('/$layer/') || path.startsWith('lib/$layer/')) {
      return layerOf[layer];
    }
  }
  return null;
}

String? _getLayerFromImport(String importPath) {
  // package:kirakira/presentation/... → presentation
  if (importPath.startsWith('presentation/')) return 'presentation';
  if (importPath.startsWith('domain/')) return 'domain';
  if (importPath.startsWith('data/')) return 'data';
  if (importPath.startsWith('core/')) return 'core';
  if (importPath.startsWith('services/')) return 'presentation';
  if (importPath.startsWith('widgets/')) return 'presentation';
  return null;
}
```

import 'dart:io';

void main() async {
  final initModule = await InitializationModule.create();
  
  // 检查所有必需 overrides 存在
  final requiredOverrides = [
    'databaseProvider',
    'characterRepositoryProvider',
    'chatRepositoryProvider',
    'worldInfoRepositoryProvider',
    'personaRepositoryProvider',
    'groupRepositoryProvider',
    'tagRepositoryProvider',

| `InitializationModule` 破坏测试 Mock | 单元测试失效 | 提供 `forTesting()` 工厂方法，支持依赖注入 |
| Provider 覆盖顺序错误 | 运行时 `UnimplementedError` | 单元测试验证 overrides 完整性 |
| 循环依赖引入 | 编译失败/运行时崩溃 | `verify_architecture.dart` 作为 CI 门禁 |
| `registerLlmProviders` 时机错误 | Provider 未注册就被使用 | 在 `InitializationModule._createServices` 中最先调用 |
| 生成文件与手动修改冲突 | `providers_registry.dart` 不一致 | 标注“勿手动编辑”，CI 中强制重新生成并对比 |
