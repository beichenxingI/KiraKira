Vulgar · 艾瑞凡

关于"Vulgar"

Vulgar，庸俗。

拉丁语的原意是"属于民众的""世俗的""普通的"。我选这个词，因为它诚实。它不假装高贵，不伪装深刻，不声称自己承载什么宏大的意义。它就是它自己——粗糙的、平凡的、属于每一个活着的人的。

关于我的叙述风格

铁匠在乎明天的铁价，不是勇者的冒险。渔夫关心今年的鱼汛，不是魔王军的动向。厨子、裁缝、酒馆老板娘、流浪剑客——他们每个人都有自己的人生，自己的烦恼，自己的秘密。他们才是这个世界真正的底色。英雄只是偶然划过的流星。

这张卡里没有固定的主角，没有预设的立场，没有必须推进的主线。你不是来扮演一个预定好的角色的——你是来进入这个世界，然后自己决定要成为谁的。

铁匠铺里的汗水，市场上的讨价还价，深夜里一盏孤灯下的叹息。这些琐碎的、泥泞的、充满烟火气的瞬间，才是构成世界的方式。我给了你一个足够大的世界，和无上限的自由度。至于你要在这里做什么、成为谁、爱上谁——那是你自己的事。

这样的设计注定挑人。它适合那些愿意慢下来、沉浸进去、在细节中发现乐趣的人。适合那些不满足于被故事牵着走、而是想要自己创造故事的人。只有少部分人能真正融入这个世界。我对此心知肚明。


艾瑞凡是一片完整的异世界大陆。不是几个城镇和一张地图，是一个真正运转的世界——从海港的鱼市到内陆的矿井，从北方的针叶林到南方的橄榄园，每个地方都有自己的气候、物产、方言、和活着的人。

你不会从一个好身份开始。不会是贵族、法师、或者天赋异禀的什么人。你是个普通人，可能是学徒、流民、或者刚到城里找活干的外乡人。你的人生不会一帆风顺。前期会很难——你要找到落脚的地方，要学会怎么在这个世界里活下去，要一点一点积累你的技能、关系、和声誉。

但这种难不是惩罚，是养成。你看着自己从一个谁都不认识的陌生人，慢慢变成这个世界的一部分。你认识了铁匠，他开始愿意赊账给你。你帮过酒馆老板娘一次忙，她记得你的名字。你学会了一门手艺，有人开始找你帮忙。这些小事慢慢堆积起来，就变成了你的人生。

这是一个白描的世界。我不会用华丽的辞藻告诉你"夕阳如血""命运的齿轮开始转动"。我会告诉你炉火的温度，面包的价格，雨后泥路的气味，和一个人在陌生城市第一次找到住处时的松一口气。这些细节构成了艾瑞凡的质感。

如果你喜欢这样的故事——缓慢的、真实的、需要沉浸的故事——欢迎来到艾瑞凡。

这里没有勇者，没有圣人，没有命中注定的伟大冒险。

这里有你自己的故事——感谢你们玩我的作品

  var _currentPage = 0;
  var _totalPages = 6;

  function flipPage(dir) {
    var newPage = _currentPage + dir;
    if (newPage < 0 || newPage >= _totalPages) return;
    _currentPage = newPage;
    updatePage();
  }

  function goPage(index) {
    if (index < 0 || index >= _totalPages) return;
    _currentPage = index;
    updatePage();
  }

  function updatePage() {
    for (var i = 0; i < _totalPages; i++) {
      var page = document.getElementById('page-' + i);
      if (page) {
        page.classList.remove('active');
        if (i === _currentPage) {
          page.classList.add('active');
        }
      }
    }

    var dots = document.querySelectorAll('#nav-dots .nav-dot');
    dots.forEach(function(dot) {
      var idx = parseInt(dot.getAttribute('data-idx'), 10);
      dot.classList.remove('current');
      if (idx === _currentPage) {
        dot.classList.add('current');
      }
    });

    var pageIndex = document.getElementById('page-index');
    if (pageIndex) {
      pageIndex.textContent = '第 ' + (_currentPage + 1) + ' 页 / 共 ' + _totalPages + ' 页';
    }
  }

  window.__intro_ready = true;

  function clampNumber(value, min, max, fallback) {
    var num = parseInt(value, 10);
    if (isNaN(num)) return fallback;
    if (num < min) return min;
    if (num > max) return max;
    return num;
  }

    if (!input) return;
    input.value = clampNumber(input.value, 10, 90, 18);
    updatePreview();
  }

  function validateLevel() {
    var input = document.getElementById('level-input');
    if (!input) return;
    input.value = clampNumber(input.value, 1, 3, 1);
    updatePreview();
  }

  function getLockedTraits() {
    return _currentTraits.filter(function(trait) {
      return _lockedTraitNames.indexOf(trait.name) !== -1;
    });
  }

  function pickRandomTraits(count, excludeNames) {
    var excluded = Array.isArray(excludeNames) ? excludeNames.slice() : [];
    var pool = _traitPool.filter(function(item) {
      return excluded.indexOf(item.name) === -1;
    });
    var result = [];
    while (result.length < count && pool.length > 0) {
      var idx = Math.floor(Math.random() * pool.length);
      result.push(pool.splice(idx, 1)[0]);
    }
    return result;
  }

  function syncLockedTraitState() {
    _lockedTraitNames = _lockedTraitNames.filter(function(name) {
      return _currentTraits.some(function(trait) {
        return trait.name === name;
      });
    }).slice(0, 2);
  }

  function toggleTraitLock(name, shouldLock) {
    var isLocked = _lockedTraitNames.indexOf(name) !== -1;

    if (shouldLock && !isLocked) {
      if (_lockedTraitNames.length >= 2) {
        renderTraits();
        return;
      }
      _lockedTraitNames.push(name);
    }

    if (!shouldLock && isLocked) {
      _lockedTraitNames = _lockedTraitNames.filter(function(item) {
        return item !== name;
      });
    }

    renderTraits();
  }

  function renderTraits() {
    var box = document.getElementById('trait-result');
    if (!box) return;
    syncLockedTraitState();
    box.innerHTML = _currentTraits.map(function(trait, index) {
      var locked = _lockedTraitNames.indexOf(trait.name) !== -1;
      return '<div class="trait-card' + (locked ? ' locked' : '') + '">' +
        '<strong>特长 ' + (index + 1) + ' · ' + trait.name + ' <small style="font-weight:400;color:#846944;">[' + trait.type + ']</small>' + (locked ? '<span class="locked-badge">已锁定</span>' : '') + '</strong>' +
        '<span>' + trait.desc + '</span>' +
        '<label class="trait-lock">' +
          '<input type="checkbox" ' + (locked ? 'checked' : '') + ' onchange="toggleTraitLockByName(\'' + trait.name.replace(/\\/g, '\\\\').replace(/'/g, "\\'") + '\', this.checked)">' +
          '锁定该特长' +
        '</label>' +
      '</div>';
    }).join('');
  }

  function randomizeTraits() {
    var lockedTraits = getLockedTraits();
    var lockedNames = lockedTraits.map(function(item){ return item.name; });
    var nextTraits = lockedTraits.slice();
    var need = 5 - nextTraits.length;
    var rerolled = pickRandomTraits(need, lockedNames);
    _currentTraits = nextTraits.concat(rerolled);
    syncLockedTraitState();
    renderTraits();
    updatePreview();
  }

  function toggleTraitLockByName(name, checked) {
    toggleTraitLock(name, checked);
    updatePreview();
  }

  function normalizeLocationItem(item) {
    return {
      name: item.name || item.title || item.label || '',
      kingdom: item.kingdom || item.country || item.realm || item.region || '',
      type: item.type || '',
      race: item.race || item.species || item.ownerRace || ''
    };
  }

  function getAvailableLocations() {
    var source = Array.isArray(window.locations) ? window.locations : _fallbackLocations;
    return source
      .map(normalizeLocationItem)
      .filter(function(item) {
        return item.name &&
          item.race === 'human' &&
          _allowedTypes.indexOf(item.type) !== -1 &&
          _allowedKingdoms.indexOf(item.kingdom) !== -1;
      });
  }

  function getCheckedValue(name) {
    var checked = document.querySelector('input[name="' + name + '"]:checked');
    return checked ? checked.value : '';
  }

  function populateCities() {
    var select = document.getElementById('city-select');
    if (!select) return;
    var list = getAvailableLocations();

    select.innerHTML = '';

    list.forEach(function(item) {
      var option = document.createElement('option');
      option.value = item.name;
      option.textContent = item.kingdom + ' · ' + item.name;
      select.appendChild(option);
    });

    if (!list.length) {
      var fallbackOption = document.createElement('option');
      fallbackOption.value = '';
      fallbackOption.textContent = '无可用城镇';
      select.appendChild(fallbackOption);
    }
  }

  function setupRadioGroup(groupId, optionSelector) {
    var group = document.getElementById(groupId);
    if (!group) return;
    group.addEventListener('click', function(event) {
      var option = event.target.closest(optionSelector || '.radio-option');
      if (!option) return;
      group.querySelectorAll(optionSelector || '.radio-option').forEach(function(node) {
        node.classList.remove('selected');
      });
      option.classList.add('selected');
      var input = option.querySelector('input[type="radio"]');
      if (input) input.checked = true;
      if (groupId === 'race-group') {
        syncRaceCityRule();
      }
      updatePreview();
    });
  }

  function syncRaceCityRule() {
    var race = getCheckedValue('race') || '人类';
    var citySelect = document.getElementById('city-select');
    var note = document.getElementById('city-locked-note');
    if (!citySelect || !note) return;

    if (race === '人类') {
      citySelect.disabled = false;
      citySelect.style.opacity = '1';
      note.innerHTML = '当前种族为<strong>人类</strong>，可选择人类城镇作为开局地点。';
    } else {
      citySelect.disabled = true;
      citySelect.style.opacity = '0.65';
      note.innerHTML = '当前种族为<strong>' + race + '</strong>，初始城市不可选。系统将根据种族背景自动分配更合适的非人类开局区域。';
    }
  }
