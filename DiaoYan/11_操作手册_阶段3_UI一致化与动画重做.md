### 3.1 扩展 `AppTheme` 增加 Glass 变体

// 在 AppTheme 类末尾添加（第 259 行前）

/// Glass 变体主题扩展
/// 使用方式：Theme.of(context).extension<GlassThemeExtension>()!.glassCard(...)
class GlassThemeExtension extends ThemeExtension<GlassThemeExtension> {
  const GlassThemeExtension();

  @override
  GlassThemeExtension copyWith() => const GlassThemeExtension();

  @override
  GlassThemeExtension lerp(ThemeExtension<GlassThemeExtension>? other, double t) => this;

  /// Glass 卡片装饰
  Decoration glassCardDecoration({
    bool isDark = true,
    double? blur,
    double? opacity,
    BorderRadius? borderRadius,
    bool showBorder = true,
    bool showShadow = true,
  }) {
    final radius = borderRadius ?? BorderRadius.circular(DesignTokens.radiusCard);
    final effectiveOpacity = opacity ?? (isDark ? 0.62 : 0.55);
    final effectiveBlur = blur ?? (isDark ? 24 : 20);
    final tint = isDark ? DesignTokens.darkCard : DesignTokens.lightCard;
    final borderColor = isDark 
        ? Colors.white.withValues(alpha: 0.08) 
        : DesignTokens.lightDivider.withValues(alpha: 0.5);
    
    return BoxDecoration(
      color: tint.withValues(alpha: (effectiveOpacity + 0.22).clamp(0.0, 0.92)),
      borderRadius: radius,
      border: showBorder ? Border.all(color: borderColor, width: 0.8) : null,
      boxShadow: showShadow 
          ? (isDark ? DesignTokens.shadowGlass : DesignTokens.shadowLevel2)
          : null,
    );
  }

  /// Glass 顶栏装饰
  Decoration glassAppBarDecoration({bool isDark = true}) => glassCardDecoration(
    isDark: isDark,
    blur: isDark ? 22 : 20,
    opacity: isDark ? 0.60 : 0.55,
    borderRadius: BorderRadius.circular(DesignTokens.radiusFull),
  );

  /// Glass 输入框装饰
  Decoration glassInputDecoration({bool isDark = true}) => glassCardDecoration(
    isDark: isDark,
    blur: isDark ? 22 : 20,
    opacity: isDark ? 0.60 : 0.55,
    borderRadius: BorderRadius.circular(DesignTokens.radiusInput),
  );

  /// Glass 面板装饰
  Decoration glassPanelDecoration({bool isDark = true}) => glassCardDecoration(
    isDark: isDark,
    blur: isDark ? 24 : 20,
    opacity: isDark ? 0.62 : 0.55,
    borderRadius: BorderRadius.circular(DesignTokens.radiusXl),
  );
}

/// 便捷获取扩展
extension GlassTheme on ThemeData {
  GlassThemeExtension get glass => extension<GlassThemeExtension>()!;
}

// lib/presentation/widgets/common/glass_widgets.dart
/// Glass 系统组件 · 基于 AppTheme.glass 扩展
/// 替代原 GlassContainer，统一使用 DesignTokens

import 'package:flutter/material.dart';
import 'package:kirakira/presentation/theme/app_theme.dart';
import 'package:kirakira/presentation/theme/design_tokens.dart';

/// Glass 卡片：最常用的玻璃容器
class GlassCard extends StatelessWidget {
  const GlassCard({
    super.key,
    required this.child,
    this.padding,
    this.margin,
    this.onTap,
    this.borderRadius,
    this.showBorder = true,
    this.showShadow = true,
  });

  final Widget child;
  final EdgeInsetsGeometry? padding;
  final EdgeInsetsGeometry? margin;
  final VoidCallback? onTap;
  final BorderRadius? borderRadius;
  final bool showBorder;
  final bool showShadow;

  @override
  Widget build(BuildContext context) {
    final theme = Theme.of(context);
    final isDark = theme.brightness == Brightness.dark;
    final decoration = theme.glass.glassCardDecoration(
      isDark: isDark,
      borderRadius: borderRadius,
      showBorder: showBorder,
      showShadow: showShadow,
    );

    Widget content = DecoratedBox(
      decoration: decoration,
      child: padding == null ? child : Padding(padding: padding!, child: child),
    );

    if (onTap != null) {
      content = Material(
        color: Colors.transparent,
        borderRadius: borderRadius ?? BorderRadius.circular(DesignTokens.radiusCard),
        child: InkWell(
          borderRadius: borderRadius ?? BorderRadius.circular(DesignTokens.radiusCard),
          onTap: onTap,
          child: content,
        ),
      );
    }

    if (margin != null) {
      content = Padding(padding: margin!, child: content);
    }

    return content;
  }
}

    super.key,
    required this.child,
    this.padding,
    this.margin,
    this.borderRadius,
    this.showBorder = true,

    final theme = Theme.of(context);
    final isDark = theme.brightness == Brightness.dark;

    Widget content = ClipRRect(
      borderRadius: borderRadius ?? BorderRadius.circular(DesignTokens.radiusXl),

        decoration: decoration.copyWith(

### 4.2 批量替换 PowerShell 脚本

**脚本**：`scripts/replace_design_tokens.ps1`

```powershell
# scripts/replace_design_tokens.ps1
# 用法: 在项目根目录运行 .\scripts\replace_design_tokens.ps1 -DryRun
# 实际执行: .\scripts\replace_design_tokens.ps1

param(
  [switch]$DryRun,
  [string]$Root = "lib"
)

$replacements = @(
  # 圆角替换
  @{ Pattern = 'BorderRadius\.circular\(16\)'; Replacement = 'BorderRadius.circular(DesignTokens.radiusLg)'; Desc = "圆角 16 → radiusLg" }
  @{ Pattern = 'BorderRadius\.circular\(22\)'; Replacement = 'BorderRadius.circular(DesignTokens.radiusFull)'; Desc = "圆角 22 → radiusFull" }
  @{ Pattern = 'BorderRadius\.circular\(24\)'; Replacement = 'BorderRadius.circular(DesignTokens.radiusFull)'; Desc = "圆角 24 → radiusFull" }
  @{ Pattern = 'BorderRadius\.circular\(8\)'; Replacement = 'BorderRadius.circular(DesignTokens.radiusSm)'; Desc = "圆角 8 → radiusSm" }
  @{ Pattern = 'BorderRadius\.circular\(30\)'; Replacement = 'BorderRadius.circular(DesignTokens.radiusFull)'; Desc = "圆角 30 → radiusFull" }
  
  # 间距替换 (需谨慎，建议人工审核)
  @{ Pattern = 'EdgeInsets\.only\(bottom:\s*10\)'; Replacement = 'EdgeInsets.only(bottom: DesignTokens.spaceSm)'; Desc = "bottom:10 → spaceSm" }
  @{ Pattern = 'EdgeInsets\.symmetric\(horizontal:\s*28\)'; Replacement = 'EdgeInsets.symmetric(horizontal: DesignTokens.spaceLg)'; Desc = "h:28 → spaceLg" }
  @{ Pattern = 'SizedBox\(height:\s*18\)'; Replacement = 'SizedBox(height: DesignTokens.spaceMd)'; Desc = "height:18 → spaceMd" }
  @{ Pattern = 'SizedBox\(width:\s*18\)'; Replacement = 'SizedBox(width: DesignTokens.spaceMd)'; Desc = "width:18 → spaceMd" }
  
  # 字号替换
  @{ Pattern = 'fontSize:\s*13(?!\d)'; Replacement = 'fontSize: DesignTokens.fontSizeBodyMedium'; Desc = "fontSize:13 → fontSizeBodyMedium" }
  @{ Pattern = 'fontSize:\s*11(?!\d)'; Replacement = 'fontSize: DesignTokens.fontSizeSm'; Desc = "fontSize:11 → fontSizeSm" }
  @{ Pattern = 'fontSize:\s*17(?!\d)'; Replacement = 'fontSize: DesignTokens.fontSizeBodyLarge'; Desc = "fontSize:17 → fontSizeBodyLarge" }
  @{ Pattern = 'fontSize:\s*15(?!\d)'; Replacement = 'fontSize: DesignTokens.fontSizeBodyLarge'; Desc = "fontSize:15 → fontSizeBodyLarge" }
  
  # 动画曲线
  @{ Pattern = 'Curves\.linear'; Replacement = 'DesignTokens.curveStandard'; Desc = "Curves.linear → curveStandard" }
  @{ Pattern = 'Curves\.easeIn'; Replacement = 'DesignTokens.curveStandard'; Desc = "Curves.easeIn → curveStandard" }
  @{ Pattern = 'Curves\.easeOut'; Replacement = 'DesignTokens.curveStandard'; Desc = "Curves.easeOut → curveStandard" }
  @{ Pattern = 'Curves\.easeInOut'; Replacement = 'DesignTokens.curveEmphasized'; Desc = "Curves.easeInOut → curveEmphasized" }
  
  # 动画时长
  @{ Pattern = 'Duration\(milliseconds:\s*500\)'; Replacement = 'Duration(milliseconds: DesignTokens.durationMd)'; Desc = "500ms → durationMd" }
  @{ Pattern = 'Duration\(milliseconds:\s*600\)'; Replacement = 'Duration(milliseconds: DesignTokens.durationLg)'; Desc = "600ms → durationLg" }
)

Write-Host "🔍 扫描 Dart 文件..." -ForegroundColor Cyan
$files = Get-ChildItem -Path $Root -Recurse -Filter "*.dart" -File

$totalChanges = 0
foreach ($file in $files) {
  $content = Get-Content -Path $file.FullName -Raw -Encoding UTF8
  $originalContent = $content
  $fileChanges = 0
  
  foreach ($r in $replacements) {
    $regex = [regex]$r.Pattern
    $matches = $regex.Matches($content)
    if ($matches.Count -gt 0) {
      $content = $regex.Replace($content, $r.Replacement)
      $fileChanges += $matches.Count
      if (!$DryRun) {
        Write-Host "  📝 $($file.FullName): $($r.Desc) x$($matches.Count)" -ForegroundColor Yellow
      }
    }
  }
  
  if ($fileChanges -gt 0) {
    $totalChanges += $fileChanges
    if (!$DryRun) {
      Set-Content -Path $file.FullName -Value $content -Encoding UTF8
      Write-Host "  ✅ $($file.FullName): $fileChanges 处替换" -ForegroundColor Green
    } else {
      Write-Host "  🔍 [DRY-RUN] $($file.FullName): $fileChanges 处将被替换" -ForegroundColor Cyan
    }
  }
}

Write-Host "`n📊 总计: $totalChanges 处替换" -ForegroundColor ($totalChanges -gt 0 ? 'Green' : 'Gray')
if ($DryRun) { Write-Host "💡 使用不带 -DryRun 参数执行实际替换" -ForegroundColor Cyan }
```

### 5.3 页面切换动画统一

**新建文件**：`lib/presentation/router/page_transitions.dart`

```dart
// lib/presentation/router/page_transitions.dart
import 'package:flutter/material.dart';
import 'package:kirakira/presentation/theme/design_tokens.dart';

/// 统一页面转场动画
PageRouteBuilder<T> buildPageRoute<T>({
  required Widget child,
  RouteSettings? settings,
  PageTransitionType type = PageTransitionType.fadeSlide,
}) {
  switch (type) {
    case PageTransitionType.fadeSlide:
      return _FadeSlideRoute(child: child, settings: settings);
    case PageTransitionType.slideRight:
      return _SlideRightRoute(child: child, settings: settings);
    case PageTransitionType.scale:
      return _ScaleRoute(child: child, settings: settings);
    case PageTransitionType.none:
      return _NoAnimationRoute(child: child, settings: settings);
  }
}

enum PageTransitionType { fadeSlide, slideRight, scale, none }

class _FadeSlideRoute<T> extends PageRouteBuilder<T> {
  _FadeSlideRoute({required Widget child, RouteSettings? settings})
      : super(
          settings: settings,
          pageBuilder: (_, __, ___) => child,
          transitionDuration: Duration(milliseconds: DesignTokens.durationPageTransition),
          reverseTransitionDuration: Duration(milliseconds: DesignTokens.durationPageTransition),
          transitionsBuilder: (_, animation, __, child) {
            return SlideTransition(
              position: Tween<Offset>(
                begin: const Offset(0, 0.1),
                end: Offset.zero,
              ).animate(CurvedAnimation(parent: animation, curve: DesignTokens.curveEmphasized)),
              child: FadeTransition(
                opacity: CurvedAnimation(parent: animation, curve: DesignTokens.curveStandard),
                child: child,
              ),
            );
          },
        );
}

class _SlideRightRoute<T> extends PageRouteBuilder<T> {
  _SlideRightRoute({required Widget child, RouteSettings? settings})
      : super(
          settings: settings,
          pageBuilder: (_, __, ___) => child,
          transitionDuration: Duration(milliseconds: DesignTokens.durationPageTransition),
          reverseTransitionDuration: Duration(milliseconds: DesignTokens.durationPageTransition),
          transitionsBuilder: (_, animation, __, child) {
            return SlideTransition(
              position: Tween<Offset>(
                begin: const Offset(1, 0),
                end: Offset.zero,
              ).animate(CurvedAnimation(parent: animation, curve: DesignTokens.curveEmphasized)),
              child: child,
            );
          },
        );
}

class _ScaleRoute<T> extends PageRouteBuilder<T> {
  _ScaleRoute({required Widget child, RouteSettings? settings})
      : super(
          settings: settings,
          pageBuilder: (_, __, ___) => child,
          transitionDuration: Duration(milliseconds: DesignTokens.durationDialog),
          reverseTransitionDuration: Duration(milliseconds: DesignTokens.durationDialog),
          transitionsBuilder: (_, animation, __, child) {
            return ScaleTransition(
              scale: CurvedAnimation(parent: animation, curve: DesignTokens.curveSpring),
              child: FadeTransition(
                opacity: CurvedAnimation(parent: animation, curve: DesignTokens.curveStandard),
                child: child,
              ),
            );
          },
        );
}

class _NoAnimationRoute<T> extends PageRouteBuilder<T> {
  _NoAnimationRoute({required Widget child, RouteSettings? settings})
      : super(
          settings: settings,
          pageBuilder: (_, __, ___) => child,
          transitionDuration: Duration.zero,
          reverseTransitionDuration: Duration.zero,
        );
}
```

## 9. 风险控制

| 风险 | 缓解措施 |
|------|----------|
| 批量替换误伤语义不同的同值字面量 | 干运行模式人工审核；仅替换高频模式；保留 `git diff` 逐文件确认 |
| `GlassContainer` 遗漏迁移导致视觉差异 | 全局搜索 `GlassContainer` 确保零引用；保留 1 版本过渡期标注 `@Deprecated` |
| 主题切换时 Glass 变体闪烁 | `GlassThemeExtension` 无状态，仅读 `ThemeData`，无副作用 |
| WebView CSS 变量不同步 | 构建脚本强制同步；CI 检查 `design_tokens.css` 存在性 |
| 动画时长过短导致感知不连贯 | 最小 200ms；用户测试确认；必要时微调 `DesignTokens` 常量 |
