```dart
/// 将文件读取为 Base64 字符串
/// 
/// 在主 isolate 中调用会阻塞 UI，**必须**在 `compute` 或 isolate 中使用
Future<String> encodeFileToBase64(String path) async {
  final bytes = await File(path).readAsBytes();
  return base64Encode(bytes);
}

/// 同步版本（仅用于已在 isolate 中的代码）
String encodeFileToBase64Sync(String path) {
  final bytes = File(path).readAsBytesSync();
  return base64Encode(bytes);
}
```

## 3. 自动化清理脚本

**脚本**：`scripts/clean_dead_code.dart`

```dart
import 'dart:io';
import 'package:args/args.dart';

void main(List<String> args) async {
  final parser = ArgParser()
    ..addFlag('dry-run', negatable: false, help: 'Only show what would be deleted')
    ..addFlag('force', negatable: false, help: 'Skip confirmations');
  
  final results = parser.parse(args);
  final dryRun = results['dry-run'] as bool;
  final force = results['force'] as bool;
  
  final tasks = [
    CleanTask(
      name: 'Delete placeholder_screen.dart',
      paths: ['lib/presentation/screens/placeholder/placeholder_screen.dart'],
      verify: () => _grep('placeholder_screen').isEmpty,
    ),
    CleanTask(
      name: 'Delete live2d_webview.dart',
      paths: ['lib/widgets/live2d_webview.dart'],
      verify: () => _grep('live2d_webview|Live2DWebView').isEmpty,
    ),
    CleanTask(
      name: 'Delete live2d_service.dart',
      paths: ['lib/services/live2d_service.dart'],
      verify: () => _grep('live2d_service|Live2DService').where((l) => !l.contains('class Live2DService')).isEmpty,
    ),
    CleanTask(
      name: 'Delete character_view_mode.dart',
      paths: ['lib/presentation/screens/character/character_view_mode.dart'],
      verify: () => _grep('CharacterViewMode').where((l) => !l.contains('enum CharacterViewMode')).isEmpty,
    ),
    CleanTask(
      name: 'Rename 无标题-1.txt',
      paths: ['lib/presentation/widgets/无标题-1.txt'],
      verify: () => true, // 手动验证
      action: () => _renameTitlessFile(),
    ),
  ];
  
  for (final task in tasks) {
    print('\n📋 ${task.name}');
    if (!task.verify()) {
      print('   ⚠️ 验证失败，跳过');
      continue;
    }
    
    if (dryRun) {
      print('   [DRY-RUN] 将删除: ${task.paths.join(', ')}');
      continue;
    }
    
    if (!force) {
      stdout.write('   确认执行? (y/N): ');
      final confirm = stdin.readLineSync()?.trim().toLowerCase();
      if (confirm != 'y') {
        print('   ⏭️ 已跳过');
        continue;
      }
    }
    
    for (final path in task.paths) {
      if (await File(path).exists()) {
        await File(path).delete();
        print('   🗑️ 已删除: $path');
      }
    }
    if (task.action != null) await task.action!();
    print('   ✅ 完成');
  }
  
  // 最终验证
  if (!dryRun) {
    print('\n🔍 运行 flutter analyze...');
    final result = await Process.run('flutter', ['analyze'], runInShell: true);
    if (result.exitCode == 0) {
      print('✅ flutter analyze 通过');
    } else {
      print('❌ flutter analyze 失败:');
      print(result.stdout);
      print(result.stderr);
      exit(1);
    }
  }
}

List<String> _grep(String pattern) {
  final result = Process.runSync('grep', ['-r', pattern, 'lib/', '--include=*.dart'], runInShell: true);
  return result.stdout.toString().trim().split('\n').where((l) => l.isNotEmpty).toList();
}

Future<void> _renameTitlessFile() async {
  final src = File('lib/presentation/widgets/无标题-1.txt');
  final dstDir = Directory('lib/presentation/widgets/regex');
  if (!await dstDir.exists()) await dstDir.create(recursive: true);
  final dst = File('${dstDir.path}/regex_script_components.dart');
  await src.rename(dst.path);
  print('   📝 已重命名: ${dst.path}');
}

class CleanTask {
  final String name;
  final List<String> paths;
  final bool Function() verify;
  final Future<void> Function()? action;
  
  CleanTask({
    required this.name,
    required this.paths,
    required this.verify,
    this.action,
  });
}
```

**使用方式**：
```bash

dart run scripts/clean_dead_code.dart

# 强制执行（CI 用）
```
